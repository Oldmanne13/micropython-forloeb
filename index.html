from machine import Pin
from time import sleep
led = Pin(5, Pin.OUT)

    led.value(1)

    led.value(0)

from time import sleep

led = Pin("LED", Pin.OUT)

while True:
    led.toggle()
    sleep(1)

Opgaven

Her skal I lave et program der kan måle afstand med en Ultrasonic Sensor. Den ultrasoniske sensor sender ultralydspulser og måler tiden, det tager for pulsen at reflektere tilbage til sensoren. Ved at bruge hastigheden af lyd i luften kan sensoren beregne afstanden til objektet. Den ultrasoniske sensor har 4 ben: VCC, GND, TRIG og ECHO. VCC og GND forbindes til strømforsyningen, mens TRIG bruges til at sende ultralydspulser, og ECHO bruges til at modtage de reflekterede pulser.

import time
from machine import Pin
import neopixel


A = Pin(7, Pin.IN)
B = Pin(15, Pin.IN)
C = Pin(5, Pin.IN)
D = Pin(4, Pin.IN)

ok = Pin(2, Pin.IN, Pin.PULL_UP)


ring = neopixel.NeoPixel(Pin(20), 16)


res = [1, 0, 1, 0]

# Farver og ventetider
colors = [
    ((255, 0, 0), 0.15),
    ((0, 255, 0), 0.05),
    ((0, 0, 255), 0.15),
    ((0, 0, 0), 0.15)
]


def success_animation():
    for j in range(16):
        for color, delay in colors:
            ring[j] = color
            ring.write()
            time.sleep(delay)


print("Hello, ESP32-S3!")

while True:


    if ok.value() == 0:

        las = [
            A.value(),
            B.value(),
            C.value(),
            D.value()
        ]

        print("A:", las[0],
              "B:", las[1],
              "C:", las[2],
              "D:", las[3])

        if las == res:
            print("success")
            success_animation()
        else:
            print("fejl")

  
        while ok.value() == 0:
            time.sleep(0.01)

  
        time.sleep(0.05)
